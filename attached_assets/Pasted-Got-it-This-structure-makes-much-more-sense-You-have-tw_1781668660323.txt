### 1. Create the VPN Helper (Frontend)
Create this file in the comix-web directory to handle the "VPN" state.
**File:** comix-web/src/lib/vpn.ts
*(Create this new file)*
```typescript
export function getProxiedImageUrl(originalUrl: string, sourceId: string): string {
  if (!originalUrl) return "";
  
  // Check localStorage for the setting we'll create in Settings
  const isVpnEnabled = localStorage.getItem("builtin_vpn_enabled") === "true";
  
  if (!isVpnEnabled) return originalUrl;

  // The backend route you already have: /api/image
  const encodedUrl = encodeURIComponent(originalUrl);
  return `/api/image?url=${encodedUrl}&source=${sourceId}`;
}

```
### 2. Add the Toggle (Frontend)

```tsx
const [isVpnEnabled, setIsVpnEnabled] = useState(
  localStorage.getItem("builtin_vpn_enabled") === "true"
);

const toggleVpn = (val: boolean) => {
  setIsVpnEnabled(val);
  localStorage.setItem("builtin_vpn_enabled", String(val));
};

// Add this into your UI JSX:
<div className="flex items-center justify-between p-4 border rounded-lg">
  <div>
    <h3 className="font-semibold">Built-in Image VPN</h3>
    <p className="text-sm text-muted-foreground">Route images through proxy to bypass blocks.</p>
  </div>
  <Switch checked={isVpnEnabled} onCheckedChange={toggleVpn} />
</div>

```

### Why this works perfectly with your structure:
 * **Zero Backend Changes:** You don't have to touch api-server/routes/image.ts because it's already a perfect, secure proxy.
 * **Frontend Logic:** By checking localStorage inside the getProxiedImageUrl function, the Reader will instantly start using the proxy the moment the user toggles the switch in Settings.
